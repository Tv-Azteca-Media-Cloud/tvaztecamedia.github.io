¿Qué valida este script?

El script validate-catalog-devpoc.sh corre 7 checks de cobertura contra el org DevPoc usando la SF CLI (sf data query). No modifica ningún dato — es puramente de lectura. Cada check genera una línea ✅ PASS o ❌ FAIL con detalle.

Al final imprime un resumen consolidado y, si hay FAILs, lista los productos concretos que tienen huecos para que puedas actuar con precisión en lugar de buscar manualmente.

Prerequisito: SF CLI autenticado con sf org login web --alias DevPoc --instance-url https://aztecamedia--devpoc.sandbox.my.salesforce.com

Script completo — guardar como validate-catalog-devpoc.sh

Copia el script completo a un archivo local y ejecútalo. Genera un reporte consolidado con todos los checks y un resumen final. 

#!/usr/bin/env bash
# F6 — Validación de cobertura del Catálogo Digital en DevPoc
# Uso: bash validate-catalog-devpoc.sh
# Requiere: sf CLI autenticado con alias DevPoc

ORG="DevPoc"
FAIL=0; PASS=0

q() { sf data query --target-org "$ORG" --query "$1" --json 2>/dev/null; }
count() { q "$1" | python3 -c "import json,sys; print(json.load(sys.stdin)['result']['totalSize'])"; }
names() { q "$1" | python3 -c "
import json,sys
for x in json.load(sys.stdin)['result']['records']:
    print(f\"    {x.get('ProductCode',x.get('Name','')):<35} | {x.get('Name','')}\")"; }

echo ""
echo "═══════════════════════════════════════════════════════"
echo "  F6 · Validación Catálogo Digital · DevPoc · $(date +%Y-%m-%d)"
echo "═══════════════════════════════════════════════════════"
echo ""

## Check 1 — Total Product2
T=$(count "SELECT COUNT() FROM Product2 WHERE IsActive = true")
if [ "$T" -ge 70 ]; then echo "  ✅ PASS | Product2 activos: $T (≥70)"; ((PASS++))
else echo "  ❌ FAIL | Product2 activos: $T — esperado ≥70"; ((FAIL++)); fi

## Check 2 — Sin AdSpaceSpecProduct (listar, no falla)
echo ""
echo "  → Productos sin AdSpaceSpecProduct junction (revisar si son no-digitales):"
names "SELECT Id, Name, ProductCode FROM Product2
       WHERE IsActive = true
         AND Id NOT IN (SELECT ProductId FROM AdSpaceSpecProduct)
       ORDER BY Name"

## Check 3 — Sin ProductCategoryProduct
C=$(count "SELECT COUNT() FROM Product2
           WHERE IsActive = true
             AND Id NOT IN (SELECT ProductId FROM ProductCategoryProduct)")
if [ "$C" -eq 0 ]; then echo ""; echo "  ✅ PASS | Todos los productos tienen ProductCategoryProduct"; ((PASS++))
else echo ""; echo "  ❌ FAIL | $C productos sin categoría:"; ((FAIL++))
  names "SELECT Id, Name, ProductCode FROM Product2
         WHERE IsActive = true
           AND Id NOT IN (SELECT ProductId FROM ProductCategoryProduct)
         ORDER BY Name"
fi

## Check 4 — Digitales sin ProductFulfillmentScenario
D=$(count "SELECT COUNT() FROM Product2
           WHERE IsActive = true
             AND Name NOT LIKE '%Producc%' AND Name NOT LIKE '%Talento%'
             AND Name NOT LIKE '%Cobertura%' AND Name NOT LIKE '%Bundle%'
             AND Name NOT LIKE 'Media TP GAM' AND Name NOT LIKE 'Video y Display%'
             AND Id NOT IN (SELECT ProductId FROM ProductFulfillmentScenario)")
if [ "$D" -eq 0 ]; then echo "  ✅ PASS | Todos los productos digitales tienen ProductFulfillmentScenario"; ((PASS++))
else echo "  ❌ FAIL | $D digitales sin fulfillment scenario:"; ((FAIL++))
  names "SELECT Id, Name, ProductCode FROM Product2
         WHERE IsActive = true
           AND Name NOT LIKE '%Producc%' AND Name NOT LIKE '%Talento%'
           AND Name NOT LIKE '%Cobertura%' AND Name NOT LIKE '%Bundle%'
           AND Name NOT LIKE 'Media TP GAM' AND Name NOT LIKE 'Video y Display%'
           AND Id NOT IN (SELECT ProductId FROM ProductFulfillmentScenario)
         ORDER BY Name"
fi

## Check 5 — Sin nombres viejos AdSpaceSpec
O=$(count "SELECT COUNT() FROM AdSpaceSpecification
           WHERE Name LIKE 'TV azteca en vivo%'
              OR (Name LIKE 'Claro Sport%' AND Name NOT LIKE 'Claro Sports%')")
if [ "$O" -eq 0 ]; then echo "  ✅ PASS | Sin AdSpaceSpecs con nombres viejos"; ((PASS++))
else echo "  ❌ FAIL | $O AdSpaceSpecs con nombre viejo aún:"; ((FAIL++))
  names "SELECT Id, Name FROM AdSpaceSpecification
         WHERE Name LIKE 'TV azteca en vivo%'
            OR (Name LIKE 'Claro Sport%' AND Name NOT LIKE 'Claro Sports%')
         ORDER BY Name"
fi

## Check 6 — Los 6 AdUnits faltantes
AU=$(count "SELECT COUNT() FROM AdTargetSegmentValue
            WHERE Name IN (
              'DeportesPruebasPleca','Sinaloa-ADN40','US-ADN40-DAI',
              'USA-ADN40-Pop-Pop-Nota',
              'Apps-USA-App-EnVivo-USA-Series-ABCForense','AZA-Envivo-USA'
            )")
if [ "$AU" -eq 6 ]; then echo "  ✅ PASS | Los 6 AdUnits faltantes existen (6/6)"; ((PASS++))
else echo "  ❌ FAIL | Solo $AU/6 AdUnits encontrados"; ((FAIL++)); fi

## Check 7 — Estado ProductFulfillmentScenario (informativo)
echo ""
echo "  → Estado ProductFulfillmentScenario (informativo — F5.5 DRO):"
q "SELECT Status, COUNT(Id) total FROM ProductFulfillmentScenario GROUP BY Status" \
  | python3 -c "
import json,sys
for x in json.load(sys.stdin)['result']['records']:
    s=x.get('Status','?'); t=x.get('expr0',0)
    icon='✅' if s in ('Active','Published') else '⚠️ '
    print(f'    {icon}  {s:<15} {t} scenarios')"

echo ""
echo "═══════════════════════════════════════════════════════"
echo "  Resumen: $PASS PASS · $FAIL FAIL"
if [ "$FAIL" -eq 0 ]; then
  echo "  ✅ Catálogo validado — listo para F5.5 (DRO Publish) y F7 (E2E)"
else
  echo "  ❌ Corregir los $FAIL FAILs antes de proceder a DRO"
fi
echo "═══════════════════════════════════════════════════════"
echo ""
